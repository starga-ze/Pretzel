# Python dependencies for pz-inferd, installed by script/install.py into
# /opt/pretzel/venv/inferd. This file is the ONLY declaration of what the daemon needs
# at runtime — before it existed, the venv was a hand-made symlink to a developer's
# checkout and a bare machine had no way to reproduce it.
#
# Deliberately NOT the corpus-build environment. That one carries the crawl-and-embed stack
# (Chroma, kubernetes, grpc, lxml, BeautifulSoup) which inferd never imports: the daemon
# reads finished vectors out of pgvector and does not crawl, chunk or embed a corpus.
# Sharing the two cost 220MB of packages that can never be reached from main.py.

# --- CPU wheels. Do not remove this line. ---
#
# PyPI's default `torch` is the CUDA build. On a box with no GPU it installs 3.4GB of
# libraries that cannot be used — nvidia/* (2.7GB) and triton (690MB) — and its own
# libtorch_cuda.so adds a further 491MB. Retrieval runs device="cpu" explicitly
# (retrieval_service.py), and torch.cuda.is_available() is False on the appliance, so
# every byte of that is dead weight in the image and on the disk.
#
# Pointing pip at the CPU index means nvidia/* and triton are never resolved at all.
# Installed size drops from ~5.3GB to ~1.1GB.
#
# Verify after install:  <venv>/bin/python -c "import torch; print(torch.__version__)"
# It must print a "+cpu" suffix. A "+cu" suffix means this line did not take effect.
--extra-index-url https://download.pytorch.org/whl/cpu

# Pinned, not floated. The embeddings in the corpus were produced by one specific model
# under one specific stack; an unpinned upgrade that changes tokenisation or pooling
# shifts the query vector away from the corpus vectors, and nothing errors — retrieval
# just quietly gets worse. Bump these deliberately, and re-run the retrieval check.
torch==2.13.0+cpu
sentence-transformers==5.7.0
transformers==5.15.0

# psycopg 3 (not psycopg2): retrieval_service imports `psycopg`. The [binary] extra ships
# the prebuilt libpq so the venv does not need a compiler at install time.
psycopg[binary]==3.3.4

# Not functional, but not optional either: a Python daemon otherwise shows up in `ps` as
# its interpreter path, which contains no "pz" substring, so `ps aux | grep pz` — the
# first thing anyone runs to see what is up — silently omits it. See set_process_title().
setproctitle==1.3.7
